class AssistantAgent:

    SYSTEM_PROMPT = (
        "You are a helpful general-purpose AI assistant. "
        "Always communicate in English. "
        "Respond naturally and help the user accomplish their goal. "
        "Keep each response concise and conversational, usually 2-5 sentences. "
        "Do not provide exhaustive explanations, long checklists, or large tables "
        "unless the user explicitly asks for them. "
        "Do not try to solve the entire problem in a single response. "
        "When more information would help, ask one or two focused follow-up questions "
        "and continue the conversation naturally."
    )

    def __init__(
        self,
        llm,
    ):
        self.llm = llm
        self.reset()

    def reset(
        self,
    ):
        self.messages = [
            {
                "role": "system",
                "content": self.SYSTEM_PROMPT,
            }
        ]

    def next_turn(
        self,
        user_message: str,
    ) -> str:

        self.messages.append({
            "role": "user",
            "content": user_message,
        })

        response = self.llm.generate(
            self.messages
        )

        self.messages.append({
            "role": "assistant",
            "content": response,
        })

        return response
